  <name>Task 1: Migration 0031_procurement</name>
  <files>backend/alembic/versions/0031_procurement.py</files>
  <action>
Create revision '0031', down_revision '0030', following 0030_crm.py style. Tables (all BigInteger Identity PKs, created_at timestamptz default now(), updated_at where the TimestampMixin is used for comparable tables):
- vendors: name String(200) NOT NULL, account_number String(64), email String(320), phone String(40), website String(300), notes Text, archived_at timestamptz. Unique index on lower(name).
- purchase_orders: number String(20) NOT NULL UNIQUE (format 'PO-<id>' is set by the service; column nullable=False with the service filling it in the same transaction is fine, or use a generated default: choose nullable String(20) UNIQUE and let the service set it right after insert), vendor_id FK vendors NOT NULL, organization_id FK organizations NULL (the client the order is for; NULL = stock), status String(10) NOT NULL default 'draft' CHECK IN ('draft','ordered','partial','received','cancelled'), ordered_on Date, expected_on Date, vendor_reference String(100), notes Text, created_by FK users NULL, created_at, updated_at.
- purchase_order_lines: purchase_order_id FK purchase_orders ON DELETE CASCADE NOT NULL, position Integer default 0, product_id FK products NULL, description Text NOT NULL, quantity Integer NOT NULL CHECK > 0, unit_cost_cents BigInteger NOT NULL CHECK >= 0, received_quantity Integer NOT NULL default 0 CHECK (received_quantity >= 0 AND received_quantity <= quantity), serialised Boolean NOT NULL default false, billable Boolean NOT NULL default true, taxable Boolean NOT NULL default true, markup_bp Integer NULL CHECK (markup_bp IS NULL OR markup_bp BETWEEN 0 AND 100000).
- inventory_items: purchase_order_line_id FK purchase_order_lines NULL, product_id FK products NULL, description Text NOT NULL, serial_number String(120) NULL, quantity Integer NOT NULL CHECK > 0, CHECK (serial_number IS NULL OR quantity = 1), unit_cost_cents BigInteger NOT NULL CHECK >= 0, status String(10) NOT NULL default 'in_stock' CHECK IN ('in_stock','delivered','consumed'), organization_id FK organizations NULL (reserved-for or delivered-to client), asset_id FK assets NULL, delivered_on Date NULL, markup_bp Integer NULL (frozen at delivery), product_charge_id FK product_charges NULL UNIQUE, received_on Date NOT NULL, notes Text, created_at, updated_at. CHECK (status <> 'delivered' OR (organization_id IS NOT NULL AND delivered_on IS NOT NULL)). Partial unique index uq_inventory_serial on lower(serial_number) WHERE serial_number IS NOT NULL. Indexes on (status), (organization_id).
- markup_rules: product_id FK products NULL UNIQUE, category String(10) NULL UNIQUE CHECK IN ('hardware','software','license','other'), markup_bp Integer NOT NULL CHECK BETWEEN 0 AND 100000, created_at, updated_at; CHECK exactly one of product_id / category is set ((product_id IS NULL) <> (category IS NULL)).
- products: add column category String(10) NULL with CHECK (category IS NULL OR category IN (...the four)).
- settings: add default_markup_bp Integer NOT NULL server_default '0' CHECK BETWEEN 0 AND 100000.
All five new tables are staff-internal: NO RLS, but GRANT SELECT, INSERT, UPDATE ON each TO the app role (use the same APP_ROLE constant/approach as 0030) and grant sequence usage. Downgrade drops everything in reverse order.
